# Automatically load initramfs files, if found

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

[cm4]
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1

[cm5]
dtoverlay=dwc2,dr_mode=peripheral
dtoverlay=vc4-kms-dsi-ili9881-7inch,dsi1,rotation=270

[all]
# Force PCIe gen 3 (faster NVME)
dtparam=pciex1
dtparam=pciex1_gen=3

# Enable CAN
dtparam=spi=on
dtoverlay=mcp2515-can0,oscillator=16000000,interrupt=20
dtoverlay=spi-bcm2835

# Disable wireless
dtoverlay=disable-bt
dtoverlay=disable-wifi

force_eeprom_read=0
camera_auto_detect=0
display_auto_detect=0

# don't try to read HDMI eeprom
hdmi_blanking=2
hdmi_ignore_edid=0xa5000080
hdmi_ignore_cec_init=1
hdmi_ignore_cec=1

dtparam=fan_temp0=30000
dtparam=fan_temp0_hyst=10000
dtparam=fan_temp0_speed=250

# Hold the regulator on while the system is running
dtoverlay=gpio-poweroff,gpiopin=2,active_low=1

# Enable UART2 for STM32 communication
dtoverlay=uart2

# Enable UART4 for GPS communication
dtoverlay=uart4

# Enable LM75 temperature sensors
dtparam=i2c_arm=on
dtoverlay=i2c-sensor,lm75,addr=0x48
dtoverlay=i2c-sensor,lm75,addr=0x49
